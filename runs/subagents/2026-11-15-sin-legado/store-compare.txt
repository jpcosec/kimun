;; deterministic store builder (fixed tree ids, timestamp): prints nothing; writes dir given as arg
(require '[kimun.host.default :as host] '[kimun.host.fs-store :as fs] '[kimun.kernel.node :as node]
         '[kimun.kernel.store :as store])
(def h host/host)
(def TS "2026-08-29T12:00:00.000Z")
(defn tid [i] (format "01ARZ3NDEKTSV4RRFFQ69G%04d" i))
(defn entry [path v] {:path path :fingerprint (str "sha-256:" v) :class "C" :projection-hash (str "projection-" v) :content-hash (str "content-" v)})
(defn ops [i]
  (let [root (node/make h :sign :block {:format :markdown :type :document :attrs {:n i}})
        ext (node/make h :sign :external {:locator {:kind :file :path (str "d" i ".md")} :sample ""
                                          :fingerprint "sha-256:1111111111111111111111111111111111111111111111111111111111111111"})]
    [{:op :add-node :node root}
     {:op :register-tree :tree {:kind :document :name (str "d" i) :class "C" :content-hash (str "content-" i) :cached? true :root (:id root)}
      :id (tid i) :root (:id root) :merkle-root (str "projection-" i) :route (entry (str "d" i ".md") i)}
     {:op :add-node :node ext}
     {:op :add-edge :edge {:type :reference :from (:id root) :to (:id ext) :evidence {:ref-hash (:id ext) :actor "jp"}}}]))
(defn plan [base os] {:plan/version 1 :base base :actor "jp" :engines {} :timestamp TS :ops (vec os)})
(def sdir (first *command-line-args*))
