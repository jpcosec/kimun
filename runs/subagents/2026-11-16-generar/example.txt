  (str "# Decisiones\n\n"
       (apply str (for [p ((:about ask) 'decision)
                        :let [t (:term (:other p))
                              src (first (for [q ((:about ask) t) :when (:anchor (:other q))] (:other q)))]]
                    (str "- " t (when src (str ": " ((:text ask) src))) "\n")))))
